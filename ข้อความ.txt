        function buildDeepSpaceBackground() {
            // Background ambient stars (10,000 background particles)
            const count = 8000;
            const geometry = new THREE.BufferGeometry();
            const positions = new Float32Array(count * 3);
            const colors = new Float32Array(count * 3);

            const colorOptions = [
                new THREE.Color(0x00f3ff), // Cyan
                new THREE.Color(0xffffff), // White
                new THREE.Color(0xa855f7), // Violet
                new THREE.Color(0xfbcfe8), // Soft pink
                new THREE.Color(0xfde047)  // Yellow
            ];

            for (let i = 0; i < count; i++) {
                const u = Math.random();
                const v = Math.random();
                const theta = u * 2.0 * Math.PI;
                const phi = Math.acos(2.0 * v - 1.0);
                const r = 500 + Math.random() * 300;

                positions[i * 3] = r * Math.sin(phi) * Math.cos(theta);
                positions[i * 3 + 1] = r * Math.sin(phi) * Math.sin(theta);
                positions[i * 3 + 2] = r * Math.cos(phi);

                const c = colorOptions[Math.floor(Math.random() * colorOptions.length)];
                colors[i * 3] = c.r;
                colors[i * 3 + 1] = c.g;
                colors[i * 3 + 2] = c.b;
            }

            geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
            geometry.setAttribute('color', new THREE.BufferAttribute(colors, 3));

            const material = new THREE.PointsMaterial({
                size: 1.5,
                vertexColors: true,
                transparent: true,
                opacity: 0.8
            });

            const bgParticles = new THREE.Points(geometry, material);
            starSphereGroup.add(bgParticles);
        }

        function buildCelestialGrid() {
            // Equatorial RA/Dec Grid Lines
            const gridMat = new THREE.LineBasicMaterial({ color: 0x0284c7, transparent: true, opacity: 0.15 });
            const radius = 150;

            // Latitude rings (Declination)
            for (let dec = -60; dec <= 60; dec += 30) {
                const points = [];
                const rDec = radius * Math.cos(THREE.MathUtils.degToRad(dec));
                const y = radius * Math.sin(THREE.MathUtils.degToRad(dec));
                for (let i = 0; i <= 64; i++) {
                    const theta = (i / 64) * Math.PI * 2;
                    points.push(new THREE.Vector3(rDec * Math.cos(theta), y, rDec * Math.sin(theta)));
                }
                const geom = new THREE.BufferGeometry().setFromPoints(points);
                gridGroup.add(new THREE.Line(geom, gridMat));
            }

            // Longitude lines (Right Ascension)
            for (let ra = 0; ra < 24; ra += 3) {
                const points = [];
                const theta = (ra / 24) * Math.PI * 2;
                for (let i = 0; i <= 64; i++) {
                    const phi = (i / 64) * Math.PI - Math.PI / 2;
                    points.push(new THREE.Vector3(
                        radius * Math.cos(phi) * Math.cos(theta),
                        radius * Math.sin(phi),
                        radius * Math.cos(phi) * Math.sin(theta)
                    ));
                }
                const geom = new THREE.BufferGeometry().setFromPoints(points);
                gridGroup.add(new THREE.Line(geom, gridMat));
            }
        }

        function buildEclipticLine() {
            // Sun's Annual Path on Sky Sphere (Ecliptic tilt ~23.44 deg)
            const points = [];
            const radius = 150;
            const tilt = THREE.MathUtils.degToRad(23.44);

            for (let i = 0; i <= 128; i++) {
                const theta = (i / 128) * Math.PI * 2;
                const x = radius * Math.cos(theta);
                const y = radius * Math.sin(theta) * Math.sin(tilt);
                const z = radius * Math.sin(theta) * Math.cos(tilt);
                points.push(new THREE.Vector3(x, y, z));
            }

            const geom = new THREE.BufferGeometry().setFromPoints(points);
            const mat = new THREE.LineDashedMaterial({
                color: 0xfacc15,
                dashSize: 4,
                gapSize: 2,
                linewidth: 2,
                transparent: true,
                opacity: 0.8
            });
            const line = new THREE.Line(geom, mat);
            line.computeLineDistances();
            eclipticGroup.add(line);
        }

        function buildHorizonGround() {
            // Stargazer Ground Horizon Ring
            const radius = 140;
            const points = [];
            for (let i = 0; i <= 64; i++) {
                const theta = (i / 64) * Math.PI * 2;
                points.push(new THREE.Vector3(radius * Math.cos(theta), 0, radius * Math.sin(theta)));
            }
            const geom = new THREE.BufferGeometry().setFromPoints(points);
            const mat = new THREE.LineBasicMaterial({ color: 0x10b981, transparent: true, opacity: 0.6 });
            horizonGroup.add(new THREE.Line(geom, mat));
            horizonGroup.visible = false; // Hidden by default
        }

        function raDecToVector3(raHours, decDeg, radius = 150) {
            const raRad = (raHours / 24) * Math.PI * 2;
            const decRad = THREE.MathUtils.degToRad(decDeg);
            return new THREE.Vector3(
                radius * Math.cos(decRad) * Math.cos(raRad),
                radius * Math.sin(decRad),
                radius * Math.cos(decRad) * Math.sin(raRad)
            );
        }

        function buildZodiac3DObjects() {
            ZODIAC_DATA.forEach((zodiac) => {
                const centerPos = raDecToVector3(zodiac.ra, zodiac.dec, 150);

                // 1. Constellation Center Node (Interactive Sphere)
                const nodeGeo = new THREE.SphereGeometry(2.5, 16, 16);
                const nodeMat = new THREE.MeshBasicMaterial({
                    color: zodiac.color,
                    wireframe: true,
                    transparent: true,
                    opacity: 0.9
                });
                const nodeMesh = new THREE.Mesh(nodeGeo, nodeMat);
                nodeMesh.position.copy(centerPos);
                nodeMesh.userData = { type: 'constellation', data: zodiac };
                constLinesGroup.add(nodeMesh);
                constNodeObjects.push(nodeMesh);

                // Glow ring around constellation node
                const ringGeo = new THREE.RingGeometry(3.5, 4.2, 32);
                const ringMat = new THREE.MeshBasicMaterial({ color: zodiac.color, side: THREE.DoubleSide, transparent: true, opacity: 0.5 });
                const ringMesh = new THREE.Mesh(ringGeo, ringMat);
                ringMesh.position.copy(centerPos);
                ringMesh.lookAt(0, 0, 0);
                constLinesGroup.add(ringMesh);

                // 2. Render Stars within Constellation
                zodiac.stars.forEach((star, index) => {
                    const offset = new THREE.Vector3(
                        (Math.random() - 0.5) * 15,
                        (Math.random() - 0.5) * 15,
                        (Math.random() - 0.5) * 15
                    );
                    const starPos = centerPos.clone().add(offset);

                    const starSize = Math.max(0.8, 2.5 - star.mag * 0.4);
                    const starGeo = new THREE.SphereGeometry(starSize, 12, 12);
                    const starMat = new THREE.MeshBasicMaterial({ color: 0xffffff });
                    const starMesh = new THREE.Mesh(starGeo, starMat);
                    starMesh.position.copy(starPos);
                    starMesh.userData = { type: 'star', data: star, parentZodiac: zodiac };
                    constLinesGroup.add(starMesh);
                });

                // 3. Render Constellation Skeleton Lines
                if (zodiac.lines && zodiac.lines.length > 0) {
                    const lineMat = new THREE.LineBasicMaterial({
                        color: zodiac.color,
                        transparent: true,
                        opacity: 0.7,
                        linewidth: 2
                    });

                    zodiac.lines.forEach((segment) => {
                        const pt1 = raDecToVector3(segment[0][0], segment[0][1], 148);
                        const pt2 = raDecToVector3(segment[1][0], segment[1][1], 148);

                        const lineGeo = new THREE.BufferGeometry().setFromPoints([pt1, pt2]);
                        const lineObj = new THREE.Line(lineGeo, lineMat);
                        constLinesGroup.add(lineObj);
                    });
                }

                // 4. Deep-Sky Objects (DSO) Marker
                if (zodiac.dso && zodiac.dso.length > 0) {
                    const dsoPos = centerPos.clone().add(new THREE.Vector3(8, -8, 5));
                    const dsoGeo = new THREE.OctahedronGeometry(2, 0);
                    const dsoMat = new THREE.MeshBasicMaterial({ color: 0xa855f7, wireframe: true });
                    const dsoMesh = new THREE.Mesh(dsoGeo, dsoMat);
                    dsoMesh.position.copy(dsoPos);
                    dsoMesh.userData = { type: 'dso', name: zodiac.dso[0], zodiac: zodiac };
                    dsoGroup.add(dsoMesh);
                }
            });
        }

        function focusConstellation(id) {
            const zodiac = ZODIAC_DATA.find(z => z.id === id);
            if (!zodiac) return;

            state.activeConstellation = zodiac;
            state.exploredIds.add(id);
            updateExplorerStats();
            renderZodiacCards();
            playSound("E5");

            // Calculate Target Camera Position
            const targetPos = raDecToVector3(zodiac.ra, zodiac.dec, 150);
            const camTargetPos = raDecToVector3(zodiac.ra, zodiac.dec, 80);

            // Animate Orbit Controls smoothly
            const duration = 1200;
            const startCamPos = camera.position.clone();
            const startTarget = controls.target.clone();
            const startTime = performance.now();

            function animateZoom(now) {
                const elapsed = now - startTime;
                const progress = Math.min(elapsed / duration, 1);
                const ease = 1 - Math.pow(1 - progress, 3); // Ease out cubic

                camera.position.lerpVectors(startCamPos, camTargetPos, ease);
                controls.target.lerpVectors(startTarget, targetPos, ease);
                controls.update();

                if (progress < 1) {
                    requestAnimationFrame(animateZoom);
                } else {
                    openInfoModal(zodiac);
                }
            }
            requestAnimationFrame(animateZoom);
        }

        // Constellation Center View Detector
        function updateConstellationDetector() {
            let closest = null;
            let minDistance = Infinity;

            const centerVec = new THREE.Vector3(0, 0, -1).applyQuaternion(camera.quaternion);

            ZODIAC_DATA.forEach(z => {
                const pos = raDecToVector3(z.ra, z.dec, 150).normalize();
                const dist = centerVec.distanceTo(pos);
                if (dist < minDistance) {
                    minDistance = dist;
                    closest = z;
                }
            });

            if (closest && minDistance < 0.8) {
                const confidence = Math.max(0, Math.round((1 - minDistance / 0.8) * 100));
                document.getElementById('detected-constellation').innerText = `${closest.nameTh} (${closest.nameEn})`;
                document.getElementById('detector-bar').style.width = `${confidence}%`;
            } else {
                document.getElementById('detected-constellation').innerText = 'หันกล้องสำรวจ...';
                document.getElementById('detector-bar').style.width = '0%';
            }
        }

        // Interactive Quiz Logic
        let currentQuizAnswer = null;
        function generateQuizQuestion() {
            const randomZodiac = ZODIAC_DATA[Math.floor(Math.random() * ZODIAC_DATA.length)];
            currentQuizAnswer = randomZodiac.id;

            document.getElementById('quiz-question').innerText = `กลุ่มดาวที่มีสัญลักษณ์ "${randomZodiac.symbol}" และมีดาวประธานคือ "${randomZodiac.alphaStar}" คือกลุ่มดาวใด?`;
            document.getElementById('quiz-hint').innerText = `คำใบ้: ปรากฏเด่นชัดที่สุดใน ${randomZodiac.season}`;

            // Generate 4 Options
            const options = [randomZodiac];
            while (options.length < 4) {
                const rand = ZODIAC_DATA[Math.floor(Math.random() * ZODIAC_DATA.length)];
                if (!options.find(o => o.id === rand.id)) {
                    options.push(rand);
                }
            }
            // Shuffle
            options.sort(() => Math.random() - 0.5);

            const optionsContainer = document.getElementById('quiz-options');
            optionsContainer.innerHTML = options.map(opt => `
                <button class="w-full py-2.5 px-4 rounded-xl glass-card text-left text-xs sm:text-sm font-semibold text-slate-200 hover:text-white hover:border-purple-400 transition" onclick="answerQuiz('${opt.id}')">
                    ${opt.nameTh} (${opt.nameEn})
                </button>
            `).join('');
        }

        function answerQuiz(selectedId) {
            state.quizTotal++;
            if (selectedId === currentQuizAnswer) {
                state.quizScore++;
                playSound("C6");
                alert("ถูกต้องแม่นยำ! 🎉 คุณได้รับ 1 คะแนน");
            } else {
                playSound("C3");
                alert("ยังไม่ถูกต้อง ลองใหม่อีกครั้งนะ!");
            }
            document.getElementById('quiz-score').innerText = `คะแนน: ${state.quizScore} / ${state.quizTotal}`;
            generateQuizQuestion();
        }

        function updateExplorerStats() {
            const pct = Math.round((state.exploredIds.size / ZODIAC_DATA.length) * 100);
            document.getElementById('stats-pct').innerText = `${pct}%`;
            document.getElementById('stats-progress-bar').style.width = `${pct}%`;
            document.getElementById('stats-const-count').innerText = `${state.exploredIds.size} / 12`;
            document.getElementById('stats-dso-count').innerText = state.dsoClickedCount;
        }

        function onCanvasClick(event) {
            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            const intersects = raycaster.intersectObjects([...constNodeObjects, ...dsoGroup.children]);

            if (intersects.length > 0) {
                const obj = intersects[0].object;
                if (obj.userData.type === 'constellation') {
                    focusConstellation(obj.userData.data.id);
                } else if (obj.userData.type === 'dso') {
                    state.dsoClickedCount++;
                    updateExplorerStats();
                    playSound("G5");
                    alert(`✨ ค้นพบวัตถุท้องฟ้าลึก (Deep-Sky Object): ${obj.userData.name}\nสังเกตได้ในกลุ่มดาว ${obj.userData.zodiac.nameTh}`);
                }
            }
        }

        function onCanvasHover(event) {
            mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
            mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

            raycaster.setFromCamera(mouse, camera);
            const intersects = raycaster.intersectObjects(constNodeObjects);

            if (intersects.length > 0) {
                document.getElementById('canvas-container').style.cursor = 'pointer';
            } else {
                document.getElementById('canvas-container').style.cursor = 'grab';
            }
        }

        function onWindowResize() {
            const container = document.getElementById('canvas-container');
            camera.aspect = container.clientWidth / container.clientHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(container.clientWidth, container.clientHeight);
        }

        function animate() {
            requestAnimationFrame(animate);

            // Auto Sky Rotation
            if (state.autoRotate) {
                starSphereGroup.rotation.y += 0.0005;
                constLinesGroup.rotation.y += 0.0005;
                dsoGroup.rotation.y += 0.0005;
            }

            // Pulse DSO Octahedrons
            const time = performance.now() * 0.002;
            dsoGroup.children.forEach(dso => {
                dso.rotation.x = time;
                dso.rotation.y = time * 0.5;
            });

            controls.update();
            updateConstellationDetector();
            renderer.render(scene, camera);
        }

            updateExplorerStats();

            // Set initial state
            controls.autoRotate = true;
            controls.autoRotateSpeed = 0.5;

            animate();
        };
    </script>
